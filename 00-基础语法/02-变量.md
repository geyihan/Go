# 00-变量的定义



变量相当于内存中一个数据存储空间的表示

# 01-变量的使用

```go
package main

import "fmt"

func main() {
	//变量的声明与赋值
	var age int = 16
    //变量的使用
	fmt.Println("age=", age)

	//变量是不可以重复定义的
	//# command-line-arguments
	//./main.go:11:6: age redeclared in this block
	//./main.go:6:6: other declaration of age
	//var age = 18
	//fmt.Println("age=", age)

	//不可以在赋值的时候给予不匹配的类型
	//# command-line-arguments
	//./main.go:18:11: undefined: init
	//var num1 init = 1.34
	var num1 = 1.34
	fmt.Println("num1=", num1)

}
```



```go
package main

import "fmt"

// 下面是全局变量，定义在函数之外的变量
var n8 = 9.8

// 全局变量可以一次性声明
var (
	name   = "han"
	age    = 23
	height = 1.84
)

func main() {
	//定义在{}之内的叫做局部变量

	// 变量的四种声明方式
	fmt.Println("----------------------------------")
	//第一种 ： 变量的使用形式
	var num int = 16
	fmt.Println(num)

	//第二种 ： 指定变量的类型，但是不赋值，使用默认值:0
	var num2 int
	fmt.Println(num2)

	//第三种 ： 不写变量类型 ，根据=后面的值自动判断类型
	var num3 = "tom"
	fmt.Println(num3)

	//第四种： 省掉var ，注意:=不能写成=
	sex := "男"
	fmt.Println(sex)

	fmt.Println("----------------------")
	// 声明多个变量
	var n1, n2, n3, n4 int
	fmt.Println(n1, n2, n3, n4)

	var name, age = "han", 23
	fmt.Println(name)
	fmt.Println(age)

	n5, height := 6.8, 183.2
	fmt.Println(n5, height)
}

```
